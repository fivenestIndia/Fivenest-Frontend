' Detect Active Document Name + Date + Time automatically
Private Function DetectDefaultJobName() As String
    On Error Resume Next
    Dim doc As Document
    Set doc = ActiveDocument
    
    Dim docBaseName As String
    docBaseName = "Jersey_Job"
    
    If Not doc Is Nothing Then
        Dim fName As String
        fName = doc.FileName
        If Len(Trim(fName)) > 0 And Not (LCase(fName) Like "untitled*") Then
            If InStrRev(fName, ".") > 0 Then
                docBaseName = Left(fName, InStrRev(fName, ".") - 1)
            Else
                docBaseName = fName
            End If
        End If
    End If
    
    Dim badChars As String, j As Long
    badChars = "\/:*?""<>|"
    For j = 1 To Len(badChars)
        docBaseName = Replace(docBaseName, Mid(badChars, j, 1), "_")
    Next j
    
    Dim dtStr As String
    dtStr = Format(Now, "yyyy-mm-dd_hh-nn")
    
    DetectDefaultJobName = docBaseName & "_" & dtStr
End Function

' Read default export folder from Windows Registry or fallback to C:\Fivenest_Export
Private Function GetDefaultExportFolder() As String
    On Error Resume Next
    Dim savedFolder As String
    savedFolder = GetSetting("FiveNest", "Settings", "DefaultFolder", "")
    If Len(Trim(savedFolder)) = 0 Then
        savedFolder = "C:\Fivenest_Export"
    End If
    GetDefaultExportFolder = savedFolder
End Function

' Browse for Folder dialog using Windows Shell
Private Function BrowseForFolder(promptText As String, initialPath As String) As String
    On Error Resume Next
    BrowseForFolder = ""
    Dim shApp As Object, fldItem As Object
    Set shApp = CreateObject("Shell.Application")
    ' 1 = BIF_RETURNONLYFSDIRS, 64 = BIF_NEWDIALOGSTYLE
    Set fldItem = shApp.BrowseForFolder(0, promptText, 65, initialPath)
    If Not fldItem Is Nothing Then
        BrowseForFolder = fldItem.Self.Path
    End If
End Function

Private Sub BrowseExportFolder()
    On Error Resume Next
    Dim curPath As String
    curPath = Trim(txtExportFolder.Text)
    If Len(curPath) = 0 Then curPath = "C:\Fivenest_Export"
    
    Dim newPath As String
    newPath = BrowseForFolder("Select FiveNest Export Folder:", curPath)
    If Len(newPath) > 0 Then
        txtExportFolder.Text = newPath
        If chkSetDefault.Value = True Then
            SaveSetting "FiveNest", "Settings", "DefaultFolder", newPath
        End If
    End If
End Sub

Private Sub RestoreTaggedShapes()
    On Error Resume Next
    Dim doc As Document
    Set doc = ActiveDocument
    If doc Is Nothing Then Exit Sub
    
    Dim p As Page
    Set p = doc.ActivePage
    If p Is Nothing Then Exit Sub
    
    Dim sh As Shape
    For Each sh In p.Shapes.All
        Select Case UCase(Trim(sh.Name))
            Case "FN_FRONT"
                MarkButtonAssigned btnFront, "[OK] Front (" & Round(sh.SizeWidth, 1) & """x" & Round(sh.SizeHeight, 1) & """)"
                GenerateThumbnail sh, imgFront, "front"
                isFrontAssigned = True
                
            Case "FN_BACK"
                MarkButtonAssigned btnBack, "[OK] Back (" & Round(sh.SizeWidth, 1) & """x" & Round(sh.SizeHeight, 1) & """)"
                GenerateThumbnail sh, imgBack, "back"
                isBackAssigned = True
                
            Case "FN_SLEEVE_LEFT_HALF"
                MarkButtonAssigned btnLeftHalf, "[OK] Left Half (" & Round(sh.SizeWidth, 1) & """x" & Round(sh.SizeHeight, 1) & """)"
                GenerateThumbnail sh, imgLeftHalf, "slv_lh"
                isLeftHalfAssigned = True
                
            Case "FN_SLEEVE_RIGHT_HALF"
                MarkButtonAssigned btnRightHalf, "[OK] Right Half (" & Round(sh.SizeWidth, 1) & """x" & Round(sh.SizeHeight, 1) & """)"
                GenerateThumbnail sh, imgRightHalf, "slv_rh"
                isRightHalfAssigned = True
                
            Case "FN_SLEEVE_LEFT_FULL"
                MarkButtonAssigned btnLeftFull, "[OK] Left Full (" & Round(sh.SizeWidth, 1) & """x" & Round(sh.SizeHeight, 1) & """)"
                GenerateThumbnail sh, imgLeftFull, "slv_lf"
                isLeftFullAssigned = True
                
            Case "FN_SLEEVE_RIGHT_FULL"
                MarkButtonAssigned btnRightFull, "[OK] Right Full (" & Round(sh.SizeWidth, 1) & """x" & Round(sh.SizeHeight, 1) & """)"
                GenerateThumbnail sh, imgRightFull, "slv_rf"
                isRightFullAssigned = True
                
            Case "FN_COLLAR"
                MarkButtonAssigned btnCollar, "[OK] Collar (" & Round(sh.SizeWidth, 1) & """x" & Round(sh.SizeHeight, 1) & """)"
                GenerateThumbnail sh, imgCollar, "collar"
                isCollarAssigned = True
        End Select
    Next sh
End Sub

' CRITICAL: Range = 2 (cdrSelection) so it exports ONLY the selected cropped shape!
Private Sub GenerateThumbnail(sh As Shape, imgControl As MSForms.Image, key As String)
    On Error Resume Next
    Dim doc As Document
    Set doc = ActiveDocument
    If doc Is Nothing Or sh Is Nothing Then Exit Sub
    
    Dim fso As Object
    Set fso = CreateObject("Scripting.FileSystemObject")
    Dim tempFolder As String
    tempFolder = fso.GetSpecialFolder(2)
    Dim thumbFile As String
    thumbFile = tempFolder & "\fn_th_" & key & "_" & Format(Now, "hhnnss") & ".jpg"
    
    doc.ClearSelection
    sh.Selected = True
    
    Dim expFltr As ExportFilter
    ' 774 = cdrJPEG, 2 = cdrSelection (ONLY selected shape cropped!), 4 = cdrRGBColorImage, 72 DPI, MaintainAspectRatio = True
    Set expFltr = doc.ExportBitmap(thumbFile, 774, 2, 4, 0, 0, 72, 72, 1, False, False, False, True, 0)
    If Not expFltr Is Nothing Then expFltr.Finish
    
    If fso.FileExists(thumbFile) Then
        imgControl.Picture = LoadPicture(thumbFile)
        imgControl.PictureSizeMode = 3 ' fmPictureSizeModeZoom (keeps true aspect ratio!)
        imgControl.BackColor = RGB(255, 255, 255)
    End If
    doc.ClearSelection
End Sub

Private Sub UpdateStatusLabel()
    Dim assignedCount As Long
    assignedCount = 0
    If isFrontAssigned Then assignedCount = assignedCount + 1
    If isBackAssigned Then assignedCount = assignedCount + 1
    If isLeftHalfAssigned Or isLeftFullAssigned Then assignedCount = assignedCount + 1
    If isRightHalfAssigned Or isRightFullAssigned Then assignedCount = assignedCount + 1
    If isCollarAssigned Then assignedCount = assignedCount + 1
    
    If isFrontAssigned And isBackAssigned And (isLeftHalfAssigned Or isLeftFullAssigned) Then
        lblStatus.Caption = "Ready! (" & assignedCount & " panels assigned). Click EXPORT below."
        lblStatus.ForeColor = RGB(38, 145, 60)
    Else
        lblStatus.Caption = assignedCount & "/4 required panels assigned (Front, Back, Sleeves)"
        lblStatus.ForeColor = RGB(120, 120, 120)
    End If
End Sub

Private Sub ExecuteExport()
    Dim doc As Document
    Set doc = ActiveDocument
    If doc Is Nothing Then
        MsgBox "Please open a jersey artwork document in CorelDRAW first.", vbExclamation, "FiveNest"
        Exit Sub
    End If
    
    Dim p As Page
    Set p = doc.ActivePage
    
    ' Find tagged shapes
    Dim shFront As Shape, shBack As Shape
    Dim shLeftSleeve As Shape, shRightSleeve As Shape
    Dim shCollar As Shape
    Dim isFullSleeve As Boolean
    
    Set shFront = FindTaggedShape(p, "FN_FRONT")
    Set shBack = FindTaggedShape(p, "FN_BACK")
    
    If shFront Is Nothing Then
        MsgBox "Please assign the FRONT panel first:" & vbCrLf & _
               "1. Select Front artwork on Corel canvas" & vbCrLf & _
               "2. Click '1. Assign Front' button", vbExclamation, "Front Panel Missing"
        Exit Sub
    End If
    If shBack Is Nothing Then
        MsgBox "Please assign the BACK panel first:" & vbCrLf & _
               "1. Select Back artwork on Corel canvas" & vbCrLf & _
               "2. Click '2. Assign Back' button", vbExclamation, "Back Panel Missing"
        Exit Sub
    End If
    
    ' Check sleeves
    If isLeftFullAssigned Or isRightFullAssigned Then
        isFullSleeve = True
        Set shLeftSleeve = FindTaggedShape(p, "FN_SLEEVE_LEFT_FULL")
        Set shRightSleeve = FindTaggedShape(p, "FN_SLEEVE_RIGHT_FULL")
    Else
        isFullSleeve = False
        Set shLeftSleeve = FindTaggedShape(p, "FN_SLEEVE_LEFT_HALF")
        Set shRightSleeve = FindTaggedShape(p, "FN_SLEEVE_RIGHT_HALF")
    End If
    
    If shLeftSleeve Is Nothing And Not shRightSleeve Is Nothing Then Set shLeftSleeve = shRightSleeve
    If shRightSleeve Is Nothing And Not shLeftSleeve Is Nothing Then Set shRightSleeve = shLeftSleeve
    
    ' Collar
    Set shCollar = FindTaggedShape(p, "FN_COLLAR")
    
    ' ================= 1. EXPORT FOLDER SELECTION =================
    If chkPromptOnExport.Value = True Then
        Dim selectedFolder As String
        selectedFolder = BrowseForFolder("Select Destination Folder for " & Trim(txtJobName.Text) & ":", Trim(txtExportFolder.Text))
        If Len(selectedFolder) = 0 Then
            ' User cancelled folder picker
            Exit Sub
        End If
        txtExportFolder.Text = selectedFolder
    End If
    
    Dim exportDir As String
    exportDir = Trim(txtExportFolder.Text)
    If Len(exportDir) = 0 Then exportDir = "C:\Fivenest_Export"
    
    ' Save default folder setting if checked
    If chkSetDefault.Value = True Then
        SaveSetting "FiveNest", "Settings", "DefaultFolder", exportDir
    End If
    SaveSetting "FiveNest", "Settings", "PromptOnExport", IIf(chkPromptOnExport.Value, "1", "0")
    
    ' ================= 2. JOB NAME & PATHS =================
    Dim jobName As String
    jobName = Trim(txtJobName.Text)
    If Len(jobName) = 0 Then jobName = DetectDefaultJobName()
    
    Dim badChars As String, k As Long
    badChars = "\/:*?""<>|"
    For k = 1 To Len(badChars)
        jobName = Replace(jobName, Mid(badChars, k, 1), "_")
    Next k
    
    Dim fso As Object, wShell As Object
    Set fso = CreateObject("Scripting.FileSystemObject")
    Set wShell = CreateObject("WScript.Shell")
    
    If Not fso.FolderExists(exportDir) Then fso.CreateFolder(exportDir)
    
    ' Clean up any legacy _Panels folder in export directory if present
    On Error Resume Next
    If fso.FolderExists(exportDir & "\" & jobName & "_Panels") Then
        fso.DeleteFolder exportDir & "\" & jobName & "_Panels", True
    End If
    On Error GoTo 0
    
    ' Use Windows system %TEMP% folder for intermediate JPGs so export folder gets ONLY the .zip file
    Dim tempSysDir As String
    tempSysDir = fso.GetSpecialFolder(2)
    Dim tempPath As String
    tempPath = tempSysDir & "\FN_Panels_" & jobName & "_" & Format(Now, "hhnnss")
    If fso.FolderExists(tempPath) Then fso.DeleteFolder tempPath, True
    fso.CreateFolder tempPath
    
    Dim count As Long
    count = 0
    Dim report As String
    report = ""
    
    ' ================= 3. EXPORT 300 DPI RGB JPGs (Range=2 cdrSelection) =================
    If ExportShape(doc, shFront, tempPath & "\front.jpg") Then
        count = count + 1
        report = report & vbCrLf & "  [OK] Front Panel (22"" x 30"")"
    End If
    If ExportShape(doc, shBack, tempPath & "\back.jpg") Then
        count = count + 1
        report = report & vbCrLf & "  [OK] Back Panel (22"" x 30"")"
    End If
    If Not shLeftSleeve Is Nothing Then
        Dim lSleeveName As String
        lSleeveName = IIf(isFullSleeve, "\full_left_sleeve.jpg", "\half_left_sleeve.jpg")
        If ExportShape(doc, shLeftSleeve, tempPath & lSleeveName) Then
            count = count + 1
            report = report & vbCrLf & "  [OK] Left Sleeve (" & IIf(isFullSleeve, "19"" x 25""", "19"" x 11""") & ")"
        End If
    End If
    If Not shRightSleeve Is Nothing Then
        Dim rSleeveName As String
        rSleeveName = IIf(isFullSleeve, "\full_right_sleeve.jpg", "\half_right_sleeve.jpg")
        If ExportShape(doc, shRightSleeve, tempPath & rSleeveName) Then
            count = count + 1
            report = report & vbCrLf & "  [OK] Right Sleeve (" & IIf(isFullSleeve, "19"" x 25""", "19"" x 11""") & ")"
        End If
    End If
    If Not shCollar Is Nothing Then
        If ExportShape(doc, shCollar, tempPath & "\collar.jpg") Then
            count = count + 1
            report = report & vbCrLf & "  [OK] Collar Band (18"" x 4.5"")"
        End If
    End If
    
    ' Write manifest.json
    Dim ts As Object
    Set ts = fso.CreateTextFile(tempPath & "\manifest.json", True)
    ts.WriteLine "{"
    ts.WriteLine "  ""version"": ""1.0"","
    ts.WriteLine "  ""jobName"": """ & jobName & ""","
    ts.WriteLine "  ""sleeveType"": """ & IIf(isFullSleeve, "full", "half") & ""","
    ts.WriteLine "  ""exportedAt"": """ & Format(Now, "yyyy-mm-dd hh:nn:ss") & """"
    ts.WriteLine "}"
    ts.Close
    
    ' ================= 4. CREATE ZIP PACKAGE IN DESTINATION FOLDER =================
    Dim zipFile As String
    zipFile = exportDir & "\" & jobName & ".zip"
    If fso.FileExists(zipFile) Then fso.DeleteFile zipFile, True
    
    Dim psCmd As String
    psCmd = "powershell.exe -NoProfile -ExecutionPolicy Bypass -Command ""Compress-Archive -Path '" & tempPath & "\*' -DestinationPath '" & zipFile & "' -Force"""
    wShell.Run psCmd, 0, True
    
    ' Delete intermediate JPG folder immediately — ONLY the .zip file remains in export folder
    On Error Resume Next

    wShell.Run "cmd /c start " & studioUrl, 0, False
    
    On Error Resume Next
    wShell.Run "explorer.exe /select,""" & zipFile & """", 1, False
    On Error GoTo 0
    
    MsgBox "FiveNest Export Complete!" & vbCrLf & vbCrLf & _

           "• OR press Ctrl+V directly on the webpage to paste it!", _
           vbInformation, "FiveNest Exporter"
           
    Unload Me
End Sub

Private Function FindTaggedShape(p As Page, tag As String) As Shape
    Set FindTaggedShape = Nothing
    On Error Resume Next
    Dim sh As Shape
    For Each sh In p.Shapes.All
        If StrComp(sh.Name, tag, vbTextCompare) = 0 Then
            Set FindTaggedShape = sh
            Exit Function
        End If
    Next sh
End Function

' CRITICAL: Range = 2 (cdrSelection) so it exports ONLY the selected cropped shape!
Private Function ExportShape(doc As Document, sh As Shape, outPath As String) As Boolean
    ExportShape = False
    On Error GoTo CatchErr
    doc.ClearSelection
    sh.Selected = True
    Dim expFltr As ExportFilter
    ' 774 = cdrJPEG, 2 = cdrSelection (ONLY selected cropped shape!), 4 = cdrRGBColorImage, 300 DPI, EmbedProfile = True
    Set expFltr = doc.ExportBitmap(outPath, 774, 2, 4, 0, 0, 300, 300, 1, False, False, True, False, 0)
    If Not expFltr Is Nothing Then expFltr.Finish
    doc.ClearSelection
    ExportShape = True
    Exit Function
CatchErr:
    ExportShape = False
End Function
